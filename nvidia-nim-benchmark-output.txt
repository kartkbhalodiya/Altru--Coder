const canvas = document.getElementById('liquidCanvas');
const ctx = canvas.getContext('2d');

function resizeCanvas() {
  canvas.width = window.innerWidth;
  canvas.height = window.innerHeight;
}

window.addEventListener('resize', resizeCanvas);
resizeCanvas();

const waves = [
  { amplitude: 50, frequency: 0.02, speed: 0.01, phase: 0 },
  { amplitude: 30, frequency: 0.03, speed: 0.015, phase: Math.PI / 2 },
  { amplitude: 70, frequency: 0.015, speed: 0.005, phase: Math.PI }
];

function animate() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  
  // Create gradient for liquid
  const gradient = ctx.createLinearGradient(0, canvas.height, 0, canvas.height / 2);
  gradient.addColorStop(0, 'rgba(102, 126, 234, 0.6)');
  gradient.addColorStop(1, 'rgba(118, 75, 162, 0.3)');
  
  ctx.fillStyle = gradient;
  
  // Draw each wave
  waves.forEach(wave => {
    wave.phase += wave.speed;
    ctx.beginPath();
    
    for (let x = 0; x < canvas.width; x += 2) {
      const y = canvas.height - wave.amplitude * Math.sin(x * wave.frequency + wave.phase);
      if (x === 0) {
        ctx.moveTo(x, y);
      } else {
        ctx.lineTo(x, y);
      }
    }
    
    ctx.lineTo(canvas.width, canvas.height);
    ctx.lineTo(0, canvas.height);
    ctx.closePath();
    ctx.fill();
  });
  
  requestAnimationFrame(animate);
}

animate();

This will create multiple overlapping waves with different amplitudes and frequencies, giving a liquid effect. The gradient makes it look more dynamic.
